import os
import sys
from pathlib import Path
from dotenv import load_dotenv
import pandas as pd
from datetime import datetime
import subprocess

# Add parent directory to path for td_client import
sys.path.insert(0, str(Path(__file__).parent))
from td_client import TrueDataClient

# Load environment variables
load_dotenv()
username = os.getenv('TRUEDATA_USERNAME')
password = os.getenv('TRUEDATA_PASSWORD')
port = int(os.getenv('TRUEDATA_PORT', 8082))

def fetch_current_prices():
    """Fetch latest prices for all portfolio stocks"""
    
    # Initialize client
    client = TrueDataClient(username, password, port)
    
    # Read portfolio holdings (from cache or Excel)
    cache_dir = Path(__file__).parent.parent / 'cache'
    portfolio_file = cache_dir / 'portfolio_holdings.csv'
    
    if not portfolio_file.exists():
        print(f"ERROR: Portfolio file not found: {portfolio_file}")
        return
    
    portfolio = pd.read_csv(portfolio_file)
    tickers = portfolio['Ticker'].unique().tolist()
    
    print(f"Fetching prices for {len(tickers)} stocks...")
    
    # Fetch prices
    prices_data = []
    for ticker in tickers:
        try:
            # Get 15-min delayed price
            data = client.get_historic_data(ticker, duration='1 D', bar_size='15 mins')
            if data is not None and not data.empty:
                latest_price = data['Close'].iloc[-1]
                latest_time = data.index[-1] if hasattr(data.index, 'tolist') else datetime.now()
                
                prices_data.append({
                    'Ticker': ticker,
                    'Price': latest_price,
                    'Timestamp': latest_time
                })
                print(f"  ✓ {ticker}: {latest_price}")
        except Exception as e:
            print(f"  ✗ {ticker}: Error - {e}")
    
    # Save to cache
    if prices_data:
        prices_df = pd.DataFrame(prices_data)
        output_file = cache_dir / 'current_prices.csv'
        prices_df.to_csv(output_file, index=False)
        print(f"\n✓ Saved {len(prices_data)} prices to {output_file}")
        return output_file
    else:
        print("\n✗ No prices fetched")
        return None

def push_to_github(cache_file):
    """Commit and push cache file to GitHub"""
    
    repo_dir = Path(__file__).parent.parent
    os.chdir(repo_dir)
    
    timestamp = datetime.now().strftime("%Y-%m-%d %I:%M %p")
    
    try:
        # Git add
        subprocess.run(['git', 'add', str(cache_file)], check=True)
        
        # Git commit
        commit_msg = f"Price update: {timestamp}"
        subprocess.run(['git', 'commit', '-m', commit_msg], check=True)
        
        # Git push
        subprocess.run(['git', 'push', 'origin', 'main'], check=True)
        
        print(f"\n✓ Pushed to GitHub at {timestamp}")
        return True
    except subprocess.CalledProcessError as e:
        print(f"\n✗ Git error: {e}")
        return False

if __name__ == "__main__":
    print(f"\n{'='*60}")
    print(f"Momentum Price Refresh - {datetime.now().strftime('%Y-%m-%d %I:%M %p')}")
    print(f"{'='*60}\n")
    
    # Fetch prices
    cache_file = fetch_current_prices()
    
    # Push to GitHub if running with --push flag
    if cache_file and '--push' in sys.argv:
        push_to_github(cache_file)
    
    print(f"\n{'='*60}")
    print("Done!")
    print(f"{'='*60}\n")
